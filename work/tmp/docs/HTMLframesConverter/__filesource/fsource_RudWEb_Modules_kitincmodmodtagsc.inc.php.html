  49. * Uses tag name

  51. function showTagsContent($tag_name) {
  52.   global $rudweb_dbtype;
  53.   global $rudweb_ln;
  54.   global $rudweb_skin;
  55.   
  56.   $qc = "SELECT DISTINCT r.cnt_id as cnt_id, m.mnu_id AS mnu_id, c.cnt_alias AS cnt_alias, m.mnu_typ AS mnu_typ,";
  57.   $qc .= " c.cnt_crd AS cnt_crd, c.cnt_crt AS cnt_crt, c.cnt_cru AS cnt_cru, c.cnt_mdd AS cnt_mdd, c.cnt_mdt AS cnt_mdt, c.cnt_mdu AS cnt_mdu";
  58.   $qc .= " FROM "._RUDWEB_PRE."ws_rel As r ";
  59.   $qc .= " LEFT JOIN "._RUDWEB_PRE."ws_tags as t ON (r.tag_id = t.tag_id)";
  60.   $qc .= " LEFT JOIN "._RUDWEB_PRE."ws_cnt as c ON (r.cnt_id = c.cnt_id)";
  61.   $qc .= " LEFT JOIN "._RUDWEB_PRE."ws_mnu as m ON (c.cnt_alias = m.mnu_cnt_alias)";
  62.   $qc .= " WHERE t.tag_name = '".$tag_name."' AND c.cnt_ln = '".$_SESSION[_RUDWEB_PRE.'usr_lang_index']."'";
  63.   $qc .= " ORDER BY c.cnt_mdd DESC, c.cnt_mdt DESC, c.cnt_crd DESC, c.cnt_crt DESC";
  64.   $rsc = rudweb_qres($rudweb_dbtype, $qc);
  65.   $t_c  = rudweb_qtotal($rudweb_dbtype, $rsc);
  66.   
  67.   if($t_c != 0) {
  68.     echo $rudweb_skin['blg']['hdr'];
  69.     for($i=1;$i<=$t_c;$i++) {
  70.       $rc = rudweb_fetch_assoc($rudweb_dbtype, $rsc);
  71.       echo $rudweb_skin['blg']['pre'];
  72.       showContent($rc['mnu_id'], $rc['cnt_alias'], $rc['mnu_typ'], 1);
  73.       echo $rudweb_skin['blg']['suf'];
  74.       if ($i !== $t_c) {echo $rudweb_skin['blg']['sep'];}
  75.     }
  76.     echo $rudweb_skin['blg']['ftr'];
  77.   }
  78. }
