  281. function check_cfg_file($path, $file) {
  282.   $err = 0;
  283.   if(!is_dir($path)) {$err = 1;} /* directory doesn't exist */
  284.   else if(!is_writable($path)) {$err = 2;} /* directory isn't writable */
  285.   else if(file_exists($path.$file)) {$err = 3;} /* file already exists */
  286.   else if(file_exists($path.$file) && !is_writable($path.$file)) {$err = 4;} /* file exists but writing is not possible */
  287.   return $err;
  288. }
  289. function write_config_str($str, $file) {
  290.   $err = 0;
  291.   $f = fopen($file, 'wb');
  292.   if($f !== false) {
  293.     $w = fwrite($f, $str);
  294.     if ($w === false) {$err = 1;} // could not write
  295.     else {$err = 0;} // write ok
  296.     fclose($f);
  297.   }
  298.   else {$err = 2;} // could not open file for writing
  299.   return $err;
  300. }

  520. function split_sql($sql, $delimiter) {
  521.   // trim
  522.   $sql = trim($sql);
  523.     // remove # comments
  524.   $sql = preg_replace('/\n{2,}/', "\n", preg_replace('/^#.*$/m', "\n", $sql));
  525.   // remove -- comments
  526.     $sql = preg_replace('/\n{2,}/', "\n", preg_replace('/^--.*$/m', "\n", $sql));
  527.   // remove \r
  528.   $sql = eregi_replace("\r", "", $sql);
  529.   // split
  530.   $sql = preg_split('/' . preg_quote($delimiter, '/') . '$/m', $sql);
  531.   // trim pieces
  532.   $sql = array_map('trim', $sql);
  533.   // delete last one if empty
  534.   $last_one = end($sql);
  535.   if(empty($last_one)) {unset($sql[key($sql)]);}
  536.   // return
  537.   return $sql;
  538. }
